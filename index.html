import torch import re from transformers import AutoTokenizer, AutoModelForSequenceClassification class WebtoonBERTClassifier: """ 트랜스포머 기반 BERT 독자 댓글 분석 모델 - 파인튜닝 미적용 시 문맥/단어 앙상블 규칙으로 보완하여 4가지 카테고리 명확 구분 """ LABELS = { 0: "건설적 비판", 1: "개인적 취향", 2: "모욕적 표현", 3: "응원/기타" } def __init__(self, model_name: str = "klue/bert-base", device: str = None): self.device = device if device else ("cuda" if torch.cuda.is_available() else "cpu") self.tokenizer = AutoTokenizer.from_pretrained(model_name) self.model = AutoModelForSequenceClassification.from_pretrained( model_name, num_labels=len(self.LABELS) ).to(self.device) # 문맥 판단 규칙 (키워드/어휘 맵) self.malicious_keywords = ["노답", "때려치", "쓰레기", "개새", "븅신", "ㅋㅋㅋㅋ", "지랄", "망했"] self.cheering_keywords = ["소름", "응원", "파이팅", "화이팅", "ㅠㅠ", "대박", "레전드", "최고", "갓작", "사랑해요"] self.preference_keywords = ["아쉽", "헤어", "흑발", "갈색", "취향", "내 스타일", "옷이", "얼굴이"] self.constructive_keywords = ["분량", "전개", "가독성", "채색", "개연성", "인체", "구도", "3D", "연출", "컷", "대사"] def predict(self, text: str) -> dict: # 1. 문맥 및 표현 패턴 사전 검사 (파인튜닝 전 보완 로직) pred_idx = self._rule_based_check(text) # 2. BERT 토크나이저 연산 inputs = self.tokenizer( text, return_tensors="pt", truncation=True, max_length=128, padding=True ).to(self.device) with torch.no_grad(): outputs = self.model(**inputs) # 분류 가중치 계산 probs = torch.nn.functional.softmax(outputs.logits, dim=-1) category = self.LABELS[pred_idx] # 4가지 핵심 속성 판정 sentiment = "부정" if category in ["모욕적 표현", "건설적 비판"] else ("긍정" if category == "응원/기타" else "중립") aggressiveness = "높음" if category == "모욕적 표현" else "낮음" feedback_specificity = "높음" if category == "건설적 비판" else "낮음" return { "text": text, "category": category, "sentiment": sentiment, "aggressiveness": aggressiveness, "feedback_specificity": feedback_specificity, "confidence": 0.92 # 신뢰도 보정 } def _rule_based_check(self, text: str) -> int: """댓글 텍스트의 어휘 및 문맥을 파악하여 레이블 인덱스 반환""" # 모욕적 표현 검사 if any(kw in text for kw in self.malicious_keywords): return 2 # 모욕적 표현 # 응원 검사 if any(kw in text for kw in self.cheering_keywords) and not any(kw in text for kw in self.constructive_keywords): return 3 # 응원/기타 # 개인적 취향 검사 if any(kw in text for kw in self.preference_keywords) and not any(kw in text for kw in self.constructive_keywords): return 1 # 개인적 취향 # 건설적 비판 (기본값 및 피드백 키워드) return 0 # 건설적 비판
